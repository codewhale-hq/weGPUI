use gpui::{prelude::*, *};
use wegpui::{
    Theme,
    accessibility::ModalScope,
    components::{
        Presence,
        dialog::{Dialog, DialogButton},
        half_screen_dialog::HalfScreenDialog,
    },
};

struct Probe {
    outer: Presence,
    inner: Presence,
    sibling: Presence,
}

impl Probe {
    fn new(window: &mut Window, cx: &mut Context<Self>) -> Self {
        let mut outer = Presence::new(cx);
        let mut sibling = Dialog::presence_for(cx);
        let mut inner = Dialog::presence_for(cx);
        outer.show_instantly(window, cx);
        sibling.show_instantly(window, cx);
        inner.show_instantly(window, cx);
        Self {
            outer,
            inner,
            sibling,
        }
    }
}

impl Render for Probe {
    fn render(&mut self, _: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
        let t = Theme::get(cx).clone();
        let page = cx.entity().downgrade();
        let close = move |which| {
            let page = page.clone();
            move |window: &mut Window, cx: &mut App| {
                page.update(cx, |this, cx| {
                    match which {
                        0 => this.outer.hide(window, cx),
                        1 => this.inner.hide(window, cx),
                        _ => this.sibling.hide(window, cx),
                    }
                    cx.notify();
                })
                .ok();
            }
        };
        let inner = Dialog::new("inner-b")
            .title("Inner B")
            .body("B is nested inside A and paints after sibling C.")
            .button(DialogButton::new("Cancel B").default_style().on_click({
                let close = close(1);
                move |window, cx| close(window, cx)
            }))
            .button(DialogButton::new("OK B").primary().on_click({
                let close = close(1);
                move |window, cx| close(window, cx)
            }))
            .presence(&self.inner)
            .on_dismiss(close(1));
        let outer = HalfScreenDialog::new("outer-a")
            .title("Outer A")
            .child(div().h(px(160.)).child("A contains B."))
            .child(inner)
            .presence(&self.outer)
            .on_dismiss(close(0));
        let sibling = Dialog::new("sibling-c")
            .title("Sibling C")
            .body("C follows A in the root layout, but paints before B.")
            .button(DialogButton::new("Close C").primary().on_click({
                let close = close(2);
                move |window, cx| close(window, cx)
            }))
            .presence(&self.sibling)
            .on_dismiss(close(2));
        ModalScope::new(
            div()
                .size_full()
                .bg(t.bg_0)
                .text_color(t.fg_0)
                .child("Page content")
                .child(outer)
                .child(sibling),
        )
    }
}

#[path = "support/platform.rs"]
mod platform;

fn main() {
    platform::application()
        .with_assets(wegpui::Assets)
        .run(|cx| {
            wegpui::init(cx);
            cx.open_window(
                WindowOptions {
                    window_bounds: Some(WindowBounds::Windowed(Bounds::new(
                        point(px(0.), px(0.)),
                        size(px(480.), px(640.)),
                    ))),
                    titlebar: Some(TitlebarOptions {
                        title: Some("weGPUI modal order probe".into()),
                        ..Default::default()
                    }),
                    ..Default::default()
                },
                |window, cx| {
                    Theme::observe(window);
                    cx.new(|cx| Probe::new(window, cx))
                },
            )
            .expect("open probe window");
            cx.activate(true);
        });
}
